from app.agents.state import AgentState

def evaluator_node(state: AgentState) -> dict:
    """
    Evaluator Node (Anti-Laziness Quality Filter).
    Inspects draft outputs from all specialist agents to ensure complete legal grounding,
    mandatory BNSS/BSA compliance, and actionable police directives.
    """
    bns_draft = state.get('bns_draft')
    bsa_draft = state.get('bsa_draft')
    cyber_draft = state.get('cyber_draft')
    iteration = state.get('iteration_count', 0)
    
    feedback = []
    
    # Check 1: Check BNS grounding
    if not bns_draft or not bns_draft.get('bns_sections'):
        feedback.append("Missing explicit BNS/IT Act penal section grounding.")
        
    # Check 2: Check BSA compliance
    if not bsa_draft or not bsa_draft.get('bsa_requirements'):
        feedback.append("Missing mandatory Section 63 BSA certificate requirement.")

    # Anti-laziness loop limit (max 2 iterations to avoid infinite loops)
    if feedback and iteration < 2:
        print(f"[!] Evaluator REJECTED drafts (Iteration {iteration + 1}): {feedback}")
        return {
            "evaluation_status": "REJECTED",
            "evaluation_feedback": feedback,
            "iteration_count": iteration + 1
        }
    else:
        print(f"[+] Evaluator APPROVED drafts! Passing to HITL review.")
        return {
            "evaluation_status": "APPROVED",
            "evaluation_feedback": [],
            "iteration_count": iteration + 1
        }

def route_after_evaluator(state: AgentState) -> str:
    """
    Conditional edge router after Evaluator node.
    If REJECTED (and under iteration limit), loops back to specialists.
    If APPROVED, proceeds to HITL Node.
    """
    status = state.get('evaluation_status', 'APPROVED')
    if status == "REJECTED":
        return "bns_specialist"
    return "hitl_node"

def build_investigation_graph():
    """
    Builds and compiles the complete Crime OS AI Agentic Investigation Graph.
    """
    workflow = StateGraph(AgentState)

    # Add Nodes
    workflow.add_node("cross_memory", cross_case_memory_node)
    workflow.add_node("bns_specialist", bns_agent_node)
    workflow.add_node("bsa_specialist", bsa_agent_node)
    workflow.add_node("cyber_specialist", cyber_agent_node)
    workflow.add_node("conventional_specialist", conventional_agent_node)
    workflow.add_node("evaluator", evaluator_node)
    workflow.add_node("hitl_node", hitl_node)
    workflow.add_node("synthesis", synthesis_node)

    # Set Entry Point
    workflow.set_entry_point("cross_memory")

    # Connect Cross-Memory to Parallel Specialist Pods
    workflow.add_edge("cross_memory", "bns_specialist")
    workflow.add_edge("cross_memory", "bsa_specialist")
    workflow.add_edge("cross_memory", "cyber_specialist")
    workflow.add_edge("cross_memory", "conventional_specialist")

    # Connect Specialist Pods to Evaluator Node
    workflow.add_edge("bns_specialist", "evaluator")
    workflow.add_edge("bsa_specialist", "evaluator")
    workflow.add_edge("cyber_specialist", "evaluator")
    workflow.add_edge("conventional_specialist", "evaluator")

    # Conditional Routing from Evaluator (Anti-laziness loop or HITL)
    workflow.add_conditional_edges(
        "evaluator",
        route_after_evaluator,
        {
            "bns_specialist": "bns_specialist",
            "hitl_node": "hitl_node"
        }
    )

    # Connect HITL to Synthesis Node
    workflow.add_edge("hitl_node", "synthesis")
    workflow.add_edge("synthesis", END)

    return workflow.compile()

investigation_graph = build_investigation_graph()

def extract_text_from_pdf(pdf_path: str) -> str:
    """
    Extracts digital text from PDF using PyMuPDF.
    """
    text = ""
    try:
        doc = fitz.open(pdf_path)
        for page in doc:
            text += page.get_text()
        doc.close()
    except Exception as e:
        print(f"[-] PyMuPDF extraction error: {e}")
    return text

import ollama
from qdrant_client import QdrantClient
from qdrant_client.models import Filter, FieldCondition, MatchValue
from config import QDRANT_HOST, QDRANT_PORT, COLLECTION_NAME

def get_qdrant_client():
    return QdrantClient(host=QDRANT_HOST, port=QDRANT_PORT)

def search_legal_sops(query: str, target_specialist: str = None, top_k: int = 5):
    """
    Searches Qdrant for legal SOPs and statutes matching the query.
    Utilizes advanced payload filtering by target_specialist.
    """
    client = get_qdrant_client()
    
    # Generate query embedding
    try:
        embed_resp = ollama.embed(model="bge-m3:latest", input=query)
        query_vector = embed_resp['embeddings'][0]
    except Exception as e:
        print(f"[-] Ollama Embedding Warning: {e}. Attempting direct query...")
        # Fallback dummy embedding if ollama local is unreachable
        query_vector = [0.0] * 1024

    # Build Qdrant Payload Filter
    query_filter = None
    if target_specialist:
        query_filter = Filter(
            must=[
                FieldCondition(
                    key="target_specialist",
                    match=MatchValue(value=target_specialist)
                )
            ]
        )

    try:
        if not client.collection_exists(COLLECTION_NAME):
            return []
            
        search_result = client.search(
            collection_name=COLLECTION_NAME,
            query_vector=query_vector,
            query_filter=query_filter,
            limit=top_k
        )
        
        results = []
        for point in search_result:
            results.append({
                "score": point.score,
                "source": point.payload.get("source", "Unknown"),
                "doc_type": point.payload.get("doc_type", "general"),
                "page": point.payload.get("page", "Unknown"),
                "text": point.payload.get("text", ""),
                "target_specialist": point.payload.get("target_specialist", "")
            })
        return results
    except Exception as e:
        print(f"[-] Qdrant Search Error: {e}")
        return []
